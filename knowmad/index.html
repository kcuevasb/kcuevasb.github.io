3// Aquí es donde Kafka dejó de ser una línea del CV.
4// Un cambio de precio en el ERP se propaga a catálogo sin
5// que ningún servicio llame a otro directamente.
6@Component
7@Slf4j
8public class ProductoEventConsumer {
9
10    private final CatalogoService catalogoService;
11
12    @KafkaListener(topics = "producto.actualizado",
13                   groupId = "catalogo-service")
14    public void onProductoActualizado(ProductoEvent event,
15                                      Acknowledgment ack) {
16        log.info("Evento recibido para SKU {}", event.sku());
17        try {
18            catalogoService.refrescar(event.sku());
19            ack.acknowledge();   // commit manual del offset
20        } catch (Exception e) {
21            log.error("Fallo procesando {} → DLQ", event.sku(), e);
22            throw e;
23        }
24    }
25}
