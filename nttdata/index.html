13@Service

15
16    private static final Logger LOG =
17            LoggerFactory.getLogger(PrescripcionService.class);
18
19    @Autowired
20    private RecetaRepository recetaRepository;
21
22    @Autowired
23    private WsSisnetClient wsSisnetClient;  // SOAP
24
25    @Transactional
26    public Receta emitir(PrescripcionDTO dto) {
27        Receta receta = new Receta(dto.getPacienteId(), dto.getFarmaco());
28        wsSisnetClient.validarColegiado(dto.getMedicoId());
29        recetaRepository.save(receta);
30        LOG.info("Receta {} emitida para paciente {}",
31                 receta.getId(), dto.getPacienteId());
32        return receta;
33    }
34}
